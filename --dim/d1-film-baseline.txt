### 关键制作代码（FFmpeg模板）

```bash
# 1) 素材顺序：black.mp4 city.mp4 eye.mp4 saltlake.mp4 earthrise.mp4 particles.mp4 heart.mp4 hands.mp4
# 2) 统一规格
for f in *.mp4; do
  ffmpeg -y -i "$f" -vf "scale=1920:1080:force_original_aspect_ratio=increase,crop=1920:1080,fps=30" \
  -c:v libx264 -pix_fmt yuv420p -an "n_$f"
done

# 3) 拼接
printf "file 'n_black.mp4'\nfile 'n_city.mp4'\nfile 'n_eye.mp4'\nfile 'n_saltlake.mp4'\nfile 'n_earthrise.mp4'\nfile 'n_particles.mp4'\nfile 'n_heart.mp4'\nfile 'n_hands.mp4'\n" > list.txt
ffmpeg -y -f concat -safe 0 -i list.txt -c copy video_noaudio.mp4

# 4) 字幕+淡入淡出+音频
ffmpeg -y -i video_noaudio.mp4 -i score.wav -filter_complex "\
[0:v]fade=t=in:st=0:d=1.2,fade=t=out:st=28.8:d=1.2,\
drawtext=fontfile=/System/Library/Fonts/PingFang.ttc:text='如果宇宙有尽头':fontcolor=white:fontsize=54:x=(w-tw)/2:y=h*0.44:alpha='if(lt(t,24),0,if(lt(t,25),t-24,1))',\
drawtext=fontfile=/System/Library/Fonts/PingFang.ttc:text='那一定是你回头的地方':fontcolor=white:fontsize=54:x=(w-tw)/2:y=h*0.54:alpha='if(lt(t,27.2),0,if(lt(t,28.2),t-27.2,1))',\
drawtext=fontfile=/System/Library/Fonts/PingFang.ttc:text='回家吧，星星':fontcolor=white:fontsize=34:x=(w-tw)/2:y=h*0.82:alpha='if(lt(t,28.3),0,if(lt(t,29),t-28.3,1))'[v];\
[1:a]afade=t=in:st=0:d=1,afade=t=out:st=27.5:d=2.5[a]" \
-map "[v]" -map "[a]" -t 30 -c:v libx264 -crf 18 -c:a aac -b:a 256k final.mp4
```
