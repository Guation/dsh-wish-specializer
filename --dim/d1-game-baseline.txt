const R=13;
let P,obs,parts,score,speed,dir,state,lastGap,spawnD,bgX,shake,dead;
let best=+(localStorage.getItem('gv')||0);

/* 音效 */
let ac=null;
function beep(f,d,ty){try{
  ac=ac||new (window.AudioContext||window.webkitAudioContext)();
  const o=ac.createOscillator(),gn=ac.createGain();
  o.type=ty||'square';o.frequency.value=f;
  gn.gain.setValueAtTime(.05,ac.currentTime);
  gn.gain.exponentialRampToValueAtTime(.0001,ac.currentTime+d);
  o.connect(gn);gn.connect(ac.destination);
  o.start();o.stop(ac.currentTime+d);
}catch(e){}}

function reset(){
  P={x:Math.min(110,W*0.22),y:H/2,vy:0};
  obs=[];parts=[];score=0;speed=280;dir=1;lastGap=null;
  spawnD=0;bgX=0;shake=0;dead=0;state='run';
}

function flip(){
  if(state==='over'){if(dead>.35)reset();return;}
  dir*=-1;P.vy*=.2;beep(520,.06);
  for(let i=0;i<9;i++){const a=Math.random()*6.283,s=80+Math.random()*260;
    parts.push({x:P.x,y:P.y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,l:.4,m:.4,c:'#8ef'});}
}
addEventListener('pointerdown',e=>{e.preventDefault();flip();});
addEventListener('keydown',e=>{
  if(['Space','ArrowUp','KeyW','ArrowDown','KeyS'].includes(e.code)){e.preventDefault();flip();}
});

function spawn(){
  const gh=Math.max(105,H*.36-score*1.8),m=46;
  let lo=m+gh/2,hi=H-m-gh/2;
  if(hi<=lo)lo=hi=H/2;
  let gy=lo+Math.random()*(hi-lo);
  if(lastGap!==null)for(let i=0;i<12;i++){
    gy=lo+Math.random()*(hi-lo);
    if(Math.abs(gy-lastGap)>H*.3)break;
  }
  lastGap=gy;
  obs.push({x:W+50,w:54,gy,gh,p:0});
  spawnD=Math.max(210,W*.55-score*4);
}

function die(){
  state='over';shake=1;dead=0;beep(110,.35,'sawtooth');
  for(let i=0;i<44;i++){const a=Math.random()*6.283,s=100+Math.random()*520;
    parts.push({x:P.x,y:P.y,vx:Math.cos(a)*s,vy:Math.sin(a)*s,
      l:.9,m:.9,c:i%2?'#ff5470':'#ffd166'});}
  if(score>best){best=score;localStorage.setItem('gv',best);}
}

function update(dt){
  shake=Math.max(0,shake-dt*3.5);
  const pv=p=>{p.x+=p.vx*dt;p.y+=p.vy*dt;p.vx*=.985;p.l-=dt;};
  if(state==='over'){
    dead+=dt;
    parts.forEach(p=>{pv(p);p.vy+=900*dt;});
    parts=parts.filter(p=>p.l>0);return;
  }
  bgX=(bgX-speed*dt)%80;
  speed=Math.min(760,280+score*7);
  P.vy+=dir*2800*dt;
  P.vy=Math.max(-1200,Math.min(1200,P.vy));
  P.y+=P.vy*dt;
  if(P.y>H-R){P.y=H-R;P.vy=0;}
  if(P.y<R){P.y=R;P.vy=0;}

  if((spawnD-=speed*dt)<=0)spawn();
  for(const o of obs){
    o.x-=speed*dt;
    if(!o.p&&o.x+o.w<P.x-R){o.p=1;score++;beep(760+Math.min(score,20)*22,.05,'triangle');}
  }
  obs=obs.filter(o=>o.x+o.w>-60);
  for(const o of obs){
    if(P.x+R>o.x&&P.x-R<o.x+o.w&&(P.y-R<o.gy-o.gh/2||P.y+R>o.gy+o.gh/2)){die();return;}
  }
  parts.forEach(p=>{pv(p);p.vy+=600*dt;});
  parts=parts.filter(p=>p.l>0);
}

  for(let i=0;i<H/80+1;i++){const yy=i*80;
    g.beginPath();g.moveTo(0,yy);g.lineTo(W,yy);g.stroke();}

  for(const o of obs){
    const tp=o.gy-o.gh/2,bt=o.gy+o.gh/2;
    g.fillStyle='#141a33';
    g.fillRect(o.x,0,o.w,tp);g.fillRect(o.x,bt,o.w,H-bt);
    g.shadowColor='#3ad6ff';g.shadowBlur=16;
    g.strokeStyle='#3ad6ff';g.lineWidth=2.5;
    g.beginPath();g.moveTo(o.x,tp);g.lineTo(o.x+o.w,tp);
    g.moveTo(o.x,bt);g.lineTo(o.x+o.w,bt);g.stroke();

    g.fillStyle='rgba(58,214,255,.16)';
    g.fillRect(o.x,0,3,tp);g.fillRect(o.x,bt,3,H-bt);
  }

  for(const p of parts){
    g.globalAlpha=Math.max(0,p.l/p.m);g.fillStyle=p.c;
    g.fillRect(p.x-2,p.y-2,4,4);
  }
  g.globalAlpha=1;

  if(state!=='over'){
    g.shadowColor=dir>0?'#7df':'#fd7';g.shadowBlur=22;
    g.fillStyle='#eafcff';const s=R*.95;
    g.beginPath();
    if(g.roundRect)g.roundRect(P.x-s,P.y-s,s*2,s*2,6);
    else g.rect(P.x-s,P.y-s,s*2,s*2);
    g.fill();g.shadowBlur=0;
  }
  g.restore();

  g.textAlign='left';g.textBaseline='top';
  g.font='bold 40px system-ui,sans-serif';g.fillStyle='#fff';
  g.fillText(score,24,20);
  g.font='600 13px system-ui';g.fillStyle='rgba(255,255,255,.4)';
  g.fillText('BEST '+best,26,68);

    g.fillStyle='rgba(255,255,255,.45)';

  if(state==='over'){

    g.font='bold 30px system-ui';g.fillStyle='#fff';

**玩法**：点击、按空格键，或者在触屏上点击，重力方向就会翻转；方块在上下两道钢管之间穿行。每穿过一组障碍 +1 分，速度越来越快，缝隙越来越窄。撞上钢管就碎。

**上瘾的机关**：
- 0.1 秒就能做出反应，死亡后可以立刻重开，节奏极快，因此“再来一把”的成本很低
- 速度随分数持续上涨，刚好比反应速度略快一点，就会一直觉得“下次肯定能过”
- 每次翻转都有粒子爆发、屏幕震动和音高递增的提示音，反馈非常密集
- 最高分保存在 localStorage，屏幕上永远显示「就差 3 分」
