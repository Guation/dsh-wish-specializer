<script type="module">
import * as THREE from 'three';
import { RoomEnvironment } from 'three/addons/environments/RoomEnvironment.js';

/* ---------------- 渲染器 ---------------- */
const renderer = new THREE.WebGLRenderer({ antialias: true, powerPreference: 'high-performance' });
renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
renderer.setSize(innerWidth, innerHeight);
renderer.shadowMap.enabled = true;
renderer.shadowMap.type = THREE.PCFSoftShadowMap;
renderer.toneMapping = THREE.ACESFilmicToneMapping;
renderer.toneMappingExposure = 1.12;
document.body.appendChild(renderer.domElement);

const scene = new THREE.Scene();
scene.background = new THREE.Color(0x05080f);
scene.fog = new THREE.FogExp2(0x05080f, 0.011);

const camera = new THREE.PerspectiveCamera(52, innerWidth / innerHeight, 0.1, 1200);

/* ---------------- 环境反射 ---------------- */
const pmrem = new THREE.PMREMGenerator(renderer);
scene.environment = pmrem.fromScene(new RoomEnvironment(), 0.04).texture;

/* ---------------- 灯光 ---------------- */
scene.add(new THREE.HemisphereLight(0x2a4a8a, 0x05060a, 0.9));

const key = new THREE.DirectionalLight(0xaaccff, 2.6);
key.position.set(-13, 22, 9);
key.castShadow = true;
key.shadow.mapSize.set(2048, 2048);
key.shadow.camera.near = 1;
key.shadow.camera.far = 90;
key.shadow.camera.left = -14;
key.shadow.camera.right = 14;
key.shadow.camera.top = 14;
key.shadow.camera.bottom = -14;
key.shadow.bias = -0.0007;
key.shadow.normalBias = 0.02;
scene.add(key, key.target);

const rimA = new THREE.PointLight(0x00ccff, 320, 45);
rimA.position.set(-10, 6, -10);
scene.add(rimA);

const rimB = new THREE.PointLight(0xff2a6d, 320, 45);
rimB.position.set(11, 5, 9);
scene.add(rimB);

/* ---------------- 地面 ---------------- */
const ground = new THREE.Mesh(
  new THREE.PlaneGeometry(900, 900),
  new THREE.MeshStandardMaterial({ color: 0x060a12, metalness: 0.65, roughness: 0.38 })
);
ground.rotation.x = -Math.PI / 2;
ground.receiveShadow = true;
scene.add(ground);

const grid = new THREE.GridHelper(700, 350, 0x1f6fa8, 0x0f2b40);
grid.material.transparent = true;
grid.material.opacity = 0.55;
grid.position.y = 0.012;
scene.add(grid);

/* ---------------- 材质 ---------------- */
const paint = new THREE.MeshPhysicalMaterial({
  color: 0xa1142a, metalness: 0.9, roughness: 0.22,
  clearcoat: 1, clearcoatRoughness: 0.09, envMapIntensity: 1.6
});
const carbon = new THREE.MeshStandardMaterial({ color: 0x121418, metalness: 0.75, roughness: 0.42 });
const tireMat = new THREE.MeshStandardMaterial({ color: 0x0a0a0d, metalness: 0.1, roughness: 0.92 });
const rimMat  = new THREE.MeshStandardMaterial({ color: 0xcfd6e0, metalness: 1, roughness: 0.16, envMapIntensity: 2.2 });
const headMat = new THREE.MeshStandardMaterial({ color: 0xffffff, emissive: 0xbde9ff, emissiveIntensity: 4.5, roughness: 0.25 });
const tailMat = new THREE.MeshStandardMaterial({ color: 0xff1220, emissive: 0xff0a1e, emissiveIntensity: 5, roughness: 0.25 });

/* ---------------- 车辆 ---------------- */
const car = new THREE.Group();
scene.add(car);

/* 车身：侧面轮廓挤出 */
const s = new THREE.Shape();
s.moveTo(-2.28, 0.32);
s.lineTo(-2.34, 0.50);
s.lineTo(-2.08, 0.68);
s.lineTo(-1.10, 0.74);
s.lineTo(-0.58, 1.16);
s.lineTo( 0.28, 1.34);
s.lineTo( 1.02, 1.05);
s.lineTo( 1.80, 0.96);
s.lineTo( 2.16, 0.90);
s.lineTo( 2.22, 0.36);
s.closePath();

const bodyGeo = new THREE.ExtrudeGeometry(s, {
  depth: 1.60, bevelEnabled: true, bevelSize: 0.06,
  bevelThickness: 0.06, bevelSegments: 3, curveSegments: 8
});
bodyGeo.translate(0, 0, -0.80);

const shell = new THREE.Mesh(bodyGeo, paint);
shell.rotation.y = Math.PI / 2;   // 车头指向 +Z
shell.castShadow = true;
shell.receiveShadow = true;
car.add(shell);

/* 前唇 / 扩散器 / 侧裙 */
const addPart = (geo, mat, pos, rot) => {
  const m = new THREE.Mesh(geo, mat);
  m.position.set(...pos);
  if (rot) m.rotation.set(...rot);
  m.castShadow = true;
  car.add(m);
  return m;
};

addPart(new THREE.BoxGeometry(1.74, 0.09, 0.52), carbon, [0, 0.27, 2.08]);
addPart(new THREE.BoxGeometry(1.66, 0.16, 0.46), carbon, [0, 0.31, -2.08]);
[-1, 1].forEach(k => addPart(new THREE.BoxGeometry(0.14, 0.13, 2.1), carbon, [k * 0.86, 0.31, 0]));

/* 尾翼 */

addPart(new THREE.BoxGeometry(1.74, 0.06, 0.42), carbon, [0, 1.24, -2.05], [-0.14, 0, 0]);

/* 后视镜 */
[-1, 1].forEach(k => addPart(new THREE.BoxGeometry(0.19, 0.09, 0.14), carbon, [k * 0.93, 1.02, 0.72]));

/* 灯组 */

tireGeo.rotateX(Math.PI / 2);                 // 轮轴 → Z
const rimGeo = new THREE.CylinderGeometry(0.30, 0.30, 0.315, 24);
rimGeo.rotateX(Math.PI / 2);
const spokeGeo = new THREE.BoxGeometry(0.07, 0.54, 0.31);

const wheels = [];
function addWheel(x, z, isFront) {
  const pivot = new THREE.Group();            // 转向
  pivot.position.set(x, 0.44, z);
  const spin = new THREE.Group();             // 滚动
  pivot.add(spin);

  const tire = new THREE.Mesh(tireGeo, tireMat);
  tire.castShadow = true;
  spin.add(tire);
  spin.add(new THREE.Mesh(rimGeo, rimMat));

  for (let i = 0; i < 6; i++) {
    const sp = new THREE.Mesh(spokeGeo, rimMat);
    sp.rotation.z = i * Math.PI / 3;
    spin.add(sp);
  }
  car.add(pivot);
  wheels.push({ pivot, spin, isFront });
}

addWheel(-0.93, 1.36, true);
addWheel( 0.93, -1.40, false);
addWheel(-0.93, -1.40, false);

  keys[e.code] = true;
  if (['ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight', 'Space'].includes(e.code)) e.preventDefault();
});
addEventListener('keyup', e => { keys[e.code] = false; });

/* ---------------- 驾驶物理 ---------------- */
const UP = new THREE.Vector3(0, 1, 0);
const MAX_SPEED = 44;
let speed = 0, steerInput = 0, heading = 0, wheelSpin = 0;
const spdEl = document.getElementById('spd');
let firstFrame = true;

function update(dt) {
  const throttle = keys.KeyW || keys.ArrowUp;
  const brake    = keys.KeyS || keys.ArrowDown;
  const left     = keys.KeyA || keys.ArrowLeft;
  const right    = keys.KeyD || keys.ArrowRight;
  const hand     = keys.Space;

  /* 纵向 */
  if (throttle) speed += 30 * dt;
  if (brake) speed -= (speed > 0.2 ? 55 : 16) * dt;
  if (!throttle && !brake) speed -= speed * 1.1 * dt;
  if (hand) speed -= speed * 4.0 * dt;
  speed = THREE.MathUtils.clamp(speed, -14, MAX_SPEED);

  /* 横向 */
  const targetSteer = (left ? 1 : 0) - (right ? 1 : 0);
  steerInput += (targetSteer - steerInput) * Math.min(1, dt * 7);

  const grip = THREE.MathUtils.clamp(Math.abs(speed) / 6, 0, 1);
  const speedFactor = 1 - Math.min(Math.abs(speed) / MAX_SPEED, 1) * 0.55;
  heading += steerInput * grip * Math.sign(speed || 1) * dt * 2.1 * speedFactor * (hand ? 1.4 : 1);

  car.rotation.y = heading;
  car.position.x += Math.sin(heading) * speed * dt;
  car.position.z += Math.cos(heading) * speed * dt;

  /* 车轮 */
  wheelSpin -= speed * dt / 0.44;
  for (const w of wheels) {
    w.spin.rotation.z = wheelSpin;
    if (w.isFront) w.pivot.rotation.y = steerInput * 0.40;
  }

  /* 相机跟随 */
  const abs = Math.min(Math.abs(speed), MAX_SPEED);
  const backDist = 8.2 + abs * 0.06;
  const camHeight = 3.0 + abs * 0.022;
  const desired = new THREE.Vector3(0, camHeight, -backDist)
    .applyAxisAngle(UP, heading).add(car.position);

  if (firstFrame) { camera.position.copy(desired); firstFrame = false; }
  else camera.position.lerp(desired, 1 - Math.pow(0.0015, dt));

  camera.lookAt(new THREE.Vector3(0, 1.1, 2.4).applyAxisAngle(UP, heading).add(car.position));

  /* 阳光跟随（保证阴影不丢失） */
  key.position.set(car.position.x - 13, 22, car.position.z + 9);
  key.target.position.copy(car.position);
  key.target.updateMatrixWorld();

  /* HUD */
  spdEl.textContent = Math.round(Math.abs(speed) * 3.6);
}

/* ---------------- 循环 ---------------- */
const clock = new THREE.Clock();
function animate() {
  requestAnimationFrame(animate);
  const dt = Math.min(clock.getDelta(), 0.05);
  update(dt);
  renderer.render(scene, camera);
}
animate();

addEventListener('resize', () => {
  camera.aspect = innerWidth / innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(innerWidth, innerHeight);
});

### 驾驶与视觉玩法

上手挺简单，重点是开起来顺手，看着也爽。

- **驾驶操作**：W或↑加速，S或↓刹车并进入倒车，A/D或←/→转向，空格触发手刹甩尾。速度上限约44 m/s，倒车和刹车手感都做了区分。
- **相机跟随**：相机在车后上方，车速越快拉得越远、抬得越高，视角自然贴合驾驶节奏。
- **车辆细节**：车身用挤出的侧面轮廓生成，加上了前唇、尾翼、后视镜、灯组和底盘霓虹，车轮会随车速滚动，前轮也会跟着转向。
- **场景氛围**：深色地面加网格、雾效、冷蓝顶光配两盏紫红/青蓝轮廓灯，让车漆反射更有质感。
---

**优化建议：** 手感不在状态的话，可以改 `MAX_SPEED` 和 `update` 里的加速数字，比如把加速的30调到45会冲得更猛，手刹的4.0改小甩尾会更稳。
