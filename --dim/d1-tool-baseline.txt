```python
#!/usr/bin/env python3
# 批量重命名工具 —— 单文件，无第三方依赖
import os, re, tkinter as tk
from collections import Counter
from tkinter import ttk, filedialog, messagebox

BAD = r'[\\/:*?"<>|]'

class App:
    def __init__(self, root):
        self.root = root
        root.title("批量重命名工具")
        root.geometry("940x660")
        self.dir = tk.StringVar()
        self.pat = tk.StringVar()
        self.rep = tk.StringVar()
        self.pre = tk.StringVar()
        self.suf = tk.StringVar()
        self.start = tk.StringVar(value="1")
        self.pad = tk.StringVar(value="3")
        self.usenum = tk.BooleanVar()
        self.pos = tk.StringVar(value="后缀")
        self.case = tk.StringVar(value="不变")
        self.newext = tk.StringVar()
        self.rec = tk.BooleanVar()
        self.usere = tk.BooleanVar(value=True)
        self.files, self.rows, self.undo_stack = [], [], []
        self._build()
        for v in (self.pat, self.rep, self.pre, self.suf, self.start,
                  self.pad, self.case, self.newext, self.usenum, self.pos, self.usere):
            v.trace_add("write", lambda *a: self.preview())
        self.rec.trace_add("write", lambda *a: self.scan())

    # ---------- 界面 ----------
    def _build(self):
        f = ttk.Frame(self.root, padding=8)
        f.pack(fill="both", expand=True)
        for c in (1, 2, 3): f.columnconfigure(c, weight=1)
        f.rowconfigure(6, weight=1)

        ttk.Label(f, text="目录").grid(row=0, column=0, sticky="w")
        ttk.Entry(f, textvariable=self.dir).grid(row=0, column=1, columnspan=3, sticky="we", padx=4)
        ttk.Button(f, text="浏览…", command=self.pick).grid(row=0, column=4, sticky="we")
        ttk.Checkbutton(f, text="包含子文件夹", variable=self.rec).grid(row=0, column=5, padx=6)

        ttk.Label(f, text="查找").grid(row=1, column=0, sticky="w")
        ttk.Entry(f, textvariable=self.pat).grid(row=1, column=1, columnspan=2, sticky="we", padx=4)
        ttk.Label(f, text="替换为").grid(row=1, column=3, sticky="e")
        ttk.Entry(f, textvariable=self.rep).grid(row=1, column=4, columnspan=2, sticky="we", padx=4)
        ttk.Checkbutton(f, text="正则", variable=self.usere).grid(row=1, column=6, padx=6)
        ttk.Label(f, text="（用 () 分组、$1 或 \\1 引用）", foreground="#888").grid(
            row=2, column=1, columnspan=4, sticky="w")

        ttk.Label(f, text="前缀").grid(row=3, column=0, sticky="w")
        ttk.Entry(f, textvariable=self.pre).grid(row=3, column=1, sticky="we", padx=4)
        ttk.Label(f, text="后缀").grid(row=3, column=2, sticky="e")
        ttk.Entry(f, textvariable=self.suf).grid(row=3, column=3, sticky="we", padx=4)
        ttk.Label(f, text="新扩展名").grid(row=3, column=4, sticky="e")
        ttk.Entry(f, textvariable=self.newext, width=8).grid(row=3, column=5, sticky="we", padx=4)
        ttk.Label(f, text="大小写").grid(row=3, column=6, sticky="e")
        ttk.Combobox(f, textvariable=self.case, values=["不变", "全小写", "全大写"],
                     state="readonly", width=8).grid(row=3, column=7, sticky="we")

        ttk.Checkbutton(f, text="加序号", variable=self.usenum).grid(row=4, column=0, sticky="w")
        ttk.Label(f, text="起始").grid(row=4, column=1, sticky="e")
        ttk.Entry(f, textvariable=self.start, width=6).grid(row=4, column=2, sticky="w", padx=4)
        ttk.Label(f, text="位数").grid(row=4, column=3, sticky="e")
        ttk.Entry(f, textvariable=self.pad, width=6).grid(row=4, column=4, sticky="w", padx=4)
        ttk.Label(f, text="位置").grid(row=4, column=5, sticky="e")
        ttk.Combobox(f, textvariable=self.pos, values=["后缀", "前缀"],
                     state="readonly", width=6).grid(row=4, column=6, sticky="w", padx=4)

        bar = ttk.Frame(f); bar.grid(row=5, column=0, columnspan=8, sticky="we", pady=6)
        ttk.Button(bar, text="刷新预览", command=self.scan).pack(side="left")
        ttk.Button(bar, text="执行重命名", command=self.run).pack(side="left", padx=8)
        ttk.Button(bar, text="撤销上次", command=self.undo).pack(side="left")

        self.tree = ttk.Treeview(f, columns=("i", "old", "new", "err"), show="headings", height=14)
        for c, t, w in (("i", "#", 45), ("old", "原文件名", 330),
                        ("new", "新文件名", 330), ("err", "状态", 140)):
            self.tree.heading(c, text=t); self.tree.column(c, width=w, anchor="w")
        self.tree.tag_configure("bad", foreground="#c0392b")
        sb = ttk.Scrollbar(f, orient="vertical", command=self.tree.yview)
        self.tree.configure(yscrollcommand=sb.set)
        self.tree.grid(row=6, column=0, columnspan=8, sticky="nsew")
        sb.grid(row=6, column=8, sticky="ns")

    # ---------- 逻辑 ----------
    def pick(self):
        d = filedialog.askdirectory()
        if d: self.dir.set(d); self.scan()

    def scan(self):
        self.files = []
        d = self.dir.get()
        if d and os.path.isdir(d):
            if self.rec.get():
                for base, _, fs in os.walk(d):
                    for x in fs: self.files.append(os.path.relpath(os.path.join(base, x), d))
            else:
                self.files = [x for x in os.listdir(d) if os.path.isfile(os.path.join(d, x))]
            self.files.sort()
        self.preview()

    def target(self, name, i):
        d, n = os.path.split(name)
        base, ext = os.path.splitext(n)
        e = self.newext.get().strip()
        if e: ext = e if e.startswith(".") else "." + e
        p = self.pat.get()
        if p:
            if self.usere.get():
                try: base = re.sub(p, self.rep.get(), base)
                except re.error: pass
            else:
                base = base.replace(p, self.rep.get())
        if self.usenum.get():
            try: num = str(int(self.start.get()) + i).zfill(max(1, int(self.pad.get())))
            except ValueError: num = ""
            base = base + num if self.pos.get() == "后缀" else num + base
        base = self.pre.get() + base + self.suf.get()
        c = self.case.get()
        if c == "全小写": base = base.lower()
        elif c == "全大写": base = base.upper()
        return os.path.join(d, base + ext) if d else base + ext

    def preview(self, *_):
        self.tree.delete(*self.tree.get_children())
        self.rows = []
        d = self.dir.get()
        tgt = [self.target(f, i) for i, f in enumerate(self.files)]
        cnt = Counter(t.lower() for t in tgt)
        for i, f in enumerate(self.files):
            t, err = tgt[i], ""
            n = os.path.basename(t)
            if t == f:
                pass
            elif not os.path.splitext(n)[0].strip():
                err = "名字为空"
            elif re.search(BAD, n):
                err = "含非法字符"
            elif cnt[t.lower()] > 1:
                err = "本批重名"
            elif os.path.exists(os.path.join(d, t)):
                err = "目标已存在"
            self.rows.append((f, t, err))
            self.tree.insert("", "end", values=(i + 1, f, t, err),
                             tags=("bad",) if err else ())

    def run(self):
        self.preview()
        if not self.rows:
            return messagebox.showinfo("提示", "请先选择目录")
        bad = [r for r in self.rows if r[2]]
        if bad:
            txt = "\n".join(f"{r[0]} → {r[1]}（{r[2]}）" for r in bad[:10])
            return messagebox.showerror("存在冲突", f"共 {len(bad)} 项有问题，请调整规则：\n\n{txt}")
        todo = [r for r in self.rows if r[0] != r[1]]
        if not todo:
            return messagebox.showinfo("提示", "没有需要改名的文件")
        if not messagebox.askyesno("确认", f"将重命名 {len(todo)} 个文件，继续？"):
            return
        d, log = self.dir.get(), []
        try:
            for old, new, _ in todo:
                o, n = os.path.join(d, old), os.path.join(d, new)
                os.rename(o, n); log.append((o, n))
        except OSError as e:
            messagebox.showerror("出错", f"{e}\n已完成 {len(log)} 个，可点「撤销上次」恢复")
        self.undo_stack.append(log)
        self.scan()

    def undo(self):
        if not self.undo_stack:
            return messagebox.showinfo("提示", "没有可撤销的操作")
        log = self.undo_stack.pop()
        ok = sum(1 for o, n in reversed(log) if self._safe(n, o))
        messagebox.showinfo("撤销", f"已恢复 {ok}/{len(log)} 个文件")
        self.scan()

    @staticmethod
    def _safe(a, b):
        try: os.rename(a, b); return True
        except OSError: return False

if __name__ == "__main__":
    r = tk.Tk()
    try:
        ttk.Style().theme_use("clam")
    except tk.TclError:
        pass
    App(r)
    r.mainloop()
```

**用法要点**

- 规则按顺序叠加：查找替换 → 序号 → 前缀/后缀 → 大小写 → 新扩展名，改任意一项实时预览。
- 正则模式下用 `(.*)\.bak` / `$1` 这类分组；勾掉「正则」则是纯文本替换。
- 序号默认追加在末尾并按位数补零（`001`、`002`…），可切成前缀形式。
- 冲突（重名、非法字符、目标已存在）会以红字标出并拒绝执行，不会误伤文件。
- 「撤销上次」按相反顺序回滚，放心试错。
