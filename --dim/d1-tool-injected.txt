# ren.py — 批量重命名工具

单文件、无依赖、跨平台。默认**只预览**，`--apply` 才动手；每次执行自动落盘日志，`--undo` 整体还原；`--csv` 可把前后对照表存档比对。

```python
#!/usr/bin/env python3
"""ren.py — 批量重命名：默认预览，--apply 执行，--undo 整体撤销。"""
import argparse, csv, datetime, difflib, json, os, re, sys, unicodedata, uuid
from pathlib import Path

JOURNAL_DIR = ".ren-journal"
RESERVED = {"CON", "PRN", "AUX", "NUL"} | {f"{p}{i}" for p in ("COM", "LPT") for i in range(1, 10)}
BAD_CHARS = set('<>:"/\\|?*')


# ---------- 显示 ----------
def nkey(p):
    return [int(t) if t.isdigit() else t.casefold() for t in re.split(r"(\d+)", p.name)]


def wid(s):
    w = 0
    for ch in s:
        if unicodedata.combining(ch):
            continue
        w += 2 if unicodedata.east_asian_width(ch) in "WF" else 1
    return w


def pad(s, w):
    return s + " " * max(0, w - wid(s))


def paint(s, code, on):
    return f"\033[{code}m{s}\033[0m" if on else s


def mark(old, new, on):
    """把旧名里被删掉的部分标红、新名里加进来的部分标绿，一眼看出差异位置。"""
    if not on:
        return old, new
    sm = difflib.SequenceMatcher(None, old, new, autojunk=False)
    o, n = [], []
    for tag, i1, i2, j1, j2 in sm.get_opcodes():
        if tag == "equal":
            o.append(old[i1:i2]); n.append(new[j1:j2])
        else:
            if i2 > i1: o.append(paint(old[i1:i2], "31", True))
            if j2 > j1: n.append(paint(new[j1:j2], "32", True))
    return "".join(o), "".join(n)


def show(rows, color, title=None):
    if title:
        print(title)
    w = max([wid(r["src"].name) for r in rows] + [1])
    ok = same = bad = 0
    for r in rows:
        has_err = any(s == "error" for s, _ in r["probs"])
        has_warn = any(s == "warn" for s, _ in r["probs"])
        if r["src"].name == r["dst"].name and not r["probs"]:
            tag, code, same = "=", "90", same + 1
        elif has_err:
            tag, code, bad = "✗", "31", bad + 1
        elif has_warn:
            tag, code, ok = "!", "33", ok + 1
        else:
            tag, code, ok = "✓", "32", ok + 1
        left, right = mark(r["src"].name, r["dst"].name, color)
        line = "  " + paint(tag, code, color) + " " + left + " " * max(0, w - wid(r["src"].name)) + "  →  " + right
        if r["probs"]:
            line += "\n      " + paint("；".join(m for _, m in r["probs"]), code, color)
        print(line)
    print(f"\n共 {len(rows)} 个：{ok} 待改名 · {same} 未变 · {bad} 有问题")


# ---------- 规则 ----------
def transform(stem, ext, idx, path, a):
    name = stem
    for pair in a.replace:
        if "=" not in pair:
            sys.exit(f"--replace 需要 OLD=NEW 形式：{pair}")
        old, new = pair.split("=", 1)
        if not old:
            sys.exit("--replace 的 OLD 不能为空")
        name = name.replace(old, new, 1) if a.replace_once else name.replace(old, new)
    if a.trim:
        name = name.strip()
    if a.squeeze:
        name = re.sub(r"\s+", " ", name)
    if a.case != "keep":
        name = getattr(name, a.case)()
    if a.template:
        try:
            name = a.template.format(
                name=name, n=idx, ext=ext.lstrip("."), parent=path.parent.name,
                date=datetime.datetime.fromtimestamp(path.stat().st_mtime).strftime(a.date_format))
        except (KeyError, IndexError, ValueError) as e:
            sys.exit(f"--template 占位符无效：{e}（可用 {{name}} {{n:03d}} {{ext}} {{parent}} {{date}}）")
    name = f"{a.prefix}{name}{a.suffix}"
    return name, (a.ext if a.ext is not None else ext)


def collect(a):
    files = []
    for raw in a.paths:
        p = Path(raw)
        if p.is_file():
            files.append(p)
        elif p.is_dir():
            it = p.rglob(a.glob) if a.recursive else p.glob(a.glob)
            files.extend(sorted((f for f in it if f.is_file()), key=nkey))
        else:
            sys.exit(f"路径不存在：{p}")
    out, seen, found = [], set(), set()
    for f in files:
        if JOURNAL_DIR in f.parts:
            continue
        rp = f.absolute()
        if rp in seen:
            continue
        seen.add(rp)
        if a.no_hidden and f.name.startswith("."):
            continue
        if any(fnmatch_match(f.name, pat) for pat in a.exclude):
            continue
        out.append(f)
        found.add(f.name)
    # 若显式给了 --map 但目录里没有对应文件，直接报错而不是静默忽略
    for k in a.map:
        if k not in found:
            sys.exit(f"--map 指定的文件不存在：{k}")
    return out


def fnmatch_match(name, pat):
    import fnmatch
    return fnmatch.fnmatch(name, pat)


def norm(p, ci):
    s = os.path.normpath(str(p.absolute()))
    return s.casefold() if ci else s


def build(files, a, ci):
    rows = []
    for i, f in enumerate(files):
        idx = a.seq_start + i * a.seq_step
        if f.name.startswith(".") and f.name.count(".") == 1:
            stem, ext = f.name, ""
        else:
            stem, ext = os.path.splitext(f.name)
        if f.name in a.map:
            newname = a.map[f.name]
        else:
            ns, ne = transform(stem, ext, idx, f, a)
            newname = ns + ne
        rows.append({"src": f, "dst": f.with_name(newname), "probs": []})

    srcset = {norm(r["src"], ci) for r in rows}
    seen = {}
    for r in rows:
        n = r["dst"].name
        if not n.strip():
            r["probs"].append(("error", "新文件名为空"))
        bad = sorted(set(n) & BAD_CHARS)
        if bad:
            r["probs"].append(("error", "含非法字符 " + " ".join(bad)))
        if any(ord(c) < 32 for c in n):
            r["probs"].append(("error", "含控制字符"))
        if n != n.rstrip(" ."):
            r["probs"].append(("error", "以空格或点结尾（Windows 不允许）"))
        base = n.split(".")[0].upper()
        if base in RESERVED:
            r["probs"].append(("error", f"{base} 是系统保留名"))
        if len(n.encode()) > 255:
            r["probs"].append(("error", f"文件名 {len(n.encode())} 字节，超过 255"))
        full = str(r["dst"].absolute())
        limit = 259 if os.name == "nt" else 4095
        if len(full) > limit:
            r["probs"].append(("error", f"完整路径 {len(full)} 字符，超过上限 {limit}"))
        elif os.name == "nt" and len(full) > 240:
            r["probs"].append(("warn", f"路径已接近上限（{len(full)} 字符）"))
        k = norm(r["dst"], ci)
        if k in seen:
            r["probs"].append(("error", "与其他文件目标重名" + ("（系统大小写不敏感）" if ci else "")))
        seen[k] = r
        if r["dst"].exists() and norm(r["dst"], ci) not in srcset:
            r["probs"].append(("error", "目标文件已存在"))
        if r["src"].name != n and norm(r["src"], ci) == norm(r["dst"], ci):
            r["probs"].append(("warn", "仅大小写变化（系统可能视为同一文件）"))
    return rows


# ---------- 执行 / 撤销 ----------
def execute(pairs):
    """两阶段改名：先全部挪到临时名，再落到目标名。任何异常整体回滚。"""
    moved, placed, errs = [], [], []
    try:
        for i, (src, _) in enumerate(pairs):
            tmp = src.parent / f".ren-tmp-{uuid.uuid4().hex[:8]}-{i}"
            if tmp.exists():
                raise RuntimeError(f"临时文件已存在：{tmp}")
            os.rename(src, tmp)
            moved.append((src, tmp))
        for j, (src, _) in enumerate(moved):
            tmp, dst = moved[j][1], pairs[j][1]
            if dst.exists():
                raise RuntimeError(f"目标在执行期间被占用：{dst}")
            os.rename(tmp, dst)
            placed.append((src, dst))
        return True, None
    except Exception as e:
        for src, dst in reversed(placed):
            try:
                os.rename(dst, src)
            except Exception as e2:
                errs.append(f"回滚失败 {dst} → {src}：{e2}")
        for src, tmp in reversed(moved):
            if tmp.exists():
                try:
                    os.rename(tmp, src)
                except Exception as e2:
                    errs.append(f"回滚失败 {tmp} → {src}：{e2}")
        return False, (e, errs)


def save_journal(entries):
    d = Path.cwd() / JOURNAL_DIR
    d.mkdir(exist_ok=True)
    p = d / f"ren-{datetime.datetime.now():%Y%m%d-%H%M%S}-{uuid.uuid4().hex[:4]}.json"
    p.write_text(json.dumps({"time": datetime.datetime.now().isoformat(timespec="seconds"),
                             "cwd": str(Path.cwd()), "entries": entries},
                            ensure_ascii=False, indent=1), encoding="utf-8")
    return p


def write_csv(path, rows):
    with open(path, "w", newline="", encoding="utf-8-sig") as f:
        w = csv.writer(f)
        w.writerow(["状态", "旧路径", "新路径", "说明"])
        for r in rows:
            if any(s == "error" for s, _ in r["probs"]):
                st = "error"
            elif r["probs"]:
                st = "warn"
            elif r["src"].name == r["dst"].name:
                st = "unchanged"
            else:
                st = "ok"
            w.writerow([st, str(r["src"]), str(r["dst"]), "；".join(m for _, m in r["probs"])])


def do_undo(path, a, color):
    jp = Path(path)
    if not jp.exists():
        print(f"日志不存在：{path}")
        return 1
    j = json.loads(jp.read_text(encoding="utf-8"))
    rows = []
    for e in reversed(j["entries"]):
        cur, orig = Path(e["dst"]), Path(e["src"])
        p = []
        if not cur.exists():
            p.append(("error", "当前文件不存在，可能已被移动或删除"))
        if orig.exists():
            p.append(("error", "原位置已被别的文件占用"))
        rows.append({"src": cur, "dst": orig, "probs": p})
    if not rows:
        print("这份日志没有记录。")
        return 0
    show(rows, color, title=f"撤销日志 {jp.name}（{j.get('time', '')}）")
    if any(any(s == "error" for s, _ in r["probs"]) for r in rows):
        print(paint("\n撤销被阻止：有文件状态不符，未做任何改动。", "31", color))
        return 1
    if not a.apply:
        print(paint("\n预览：以上文件将还原为原名。加 --apply 执行撤销。", "36", color))
        return 0
    ok, err = execute([(r["src"], r["dst"]) for r in rows])
    if not ok:
        e, errs = err
        print(paint(f"\n撤销失败并已回滚：{e}", "31", color))
        for x in errs:
            print(paint("  " + x, "31", color))
        return 1
    print(paint(f"\n已撤销：{len(rows)} 个文件恢复原名。", "32", color))
    return 0


# ---------- 入口 ----------
def parse(argv=None):
    p = argparse.ArgumentParser(prog="ren.py", description="批量重命名：默认预览，--apply 执行，--undo 撤销")
    p.add_argument("paths", nargs="*", default=["."])
    p.add_argument("-r", "--recursive", action="store_true")
    p.add_argument("-g", "--glob", default="*")
    p.add_argument("-x", "--exclude", action="append", default=[])
    p.add_argument("--no-hidden", action="store_true")
    p.add_argument("--replace", action="append", default=[])
    p.add_argument("--replace-once", action="store_true")
    p.add_argument("--trim", action="store_true")
    p.add_argument("--squeeze", action="store_true")
    p.add_argument("--case", choices=["lower", "upper", "title", "keep"], default="keep")
    p.add_argument("--prefix", default="")
    p.add_argument("--suffix", default="")
    p.add_argument("--template")
    p.add_argument("--ext")
    p.add_argument("--seq-start", type=int, default=1)
    p.add_argument("--seq-step", type=int, default=1)
    p.add_argument("--date-format", default="%Y%m%d")
    p.add_argument("--map", action="append", default=[])
    p.add_argument("--apply", action="store_true")
    p.add_argument("--undo", metavar="JOURNAL")
    p.add_argument("--list-journals", action="store_true")
    p.add_argument("--csv", metavar="PATH")
    p.add_argument("--on-conflict", choices=["abort", "skip"], default="abort")
    p.add_argument("--ci-check", choices=["auto", "on", "off"], default="auto")
    p.add_argument("--no-color", action="store_true")
    a = p.parse_args(argv)
    if not a.paths:
        a.paths = ["."]
    a.map = dict(x.split("=", 1) for x in a.map) if a.map else {}
    if a.ext is not None and a.ext and not a.ext.startswith("."):
        a.ext = "." + a.ext
    return a


def main(argv=None):
    a = parse(argv)
    color = (not a.no_color) and sys.stdout.isatty()

    if a.list_journals:
        d = Path.cwd() / JOURNAL_DIR
        if not d.exists():
            print("没有历史日志。")
            return 0
        for f in sorted(d.glob("*.json")):
            j = json.loads(f.read_text(encoding="utf-8"))
            print(f"{f.name}  {j.get('time', '')}  {len(j['entries'])} 个文件")
        return 0

    if a.undo:
        return do_undo(a.undo, a, color)

    files = collect(a)
    if not files:
        print("没有匹配到文件。")
        return 0
    ci = a.ci_check == "on" or (a.ci_check == "auto" and (os.name == "nt" or sys.platform == "darwin"))
    rows = build(files, a, ci)
    show(rows, color)
    if a.csv:
        write_csv(a.csv, rows)
        print(f"对照表已写入 {a.csv}")

    bad = [r for r in rows if any(s == "error" for s, _ in r["probs"])]
    if bad and a.on_conflict == "abort":
        print(paint(f"\n发现 {len(bad)} 个冲突，已阻止执行（未改动任何文件）。"
                    f"用 --on-conflict skip 跳过它们，或调整规则。", "31", color))
        return 1

    pairs = [(r["src"], r["dst"]) for r in rows
             if r["src"].name != r["dst"].name and not any(s == "error" for s, _ in r["probs"])]
    if not pairs:
        print("\n没有需要改名的文件。")
        return 0
    if not a.apply:
        print(paint(f"\n预览：将重命名 {len(pairs)} 个文件。加 --apply 执行。", "36", color))
        return 0

    entries = []
    for src, dst in pairs:
        st = src.stat()
        entries.append({"src": str(src.absolute()), "dst": str(dst.absolute()),
                        "size": st.st_size, "mtime_ns": st.st_mtime_ns})

    ok, err = execute(pairs)
    if not ok:
        e, errs = err
        print(paint(f"\n执行失败，已整体回滚：{e}", "31", color))
        for x in errs:
            print(paint("  " + x, "31", color))
        return 1
    jp = save_journal(entries)
    print(paint(f"\n完成：{len(pairs)} 个文件已重命名。", "32", color))
    print(f"撤销：python {Path(sys.argv[0]).name} --undo {jp} --apply")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

## 用法

```bash
# 1. 先看效果（不动文件）：把 IMG_ 前缀去掉，加三位序号，统一扩展名
python ren.py ./照片 -g "*.jpeg" --replace "IMG_=" --template "{name}_{n:03d}" --ext jpg

# 2. 按拍摄日期打头，多规则叠加后的结果直接读得懂
python ren.py ./照片 --template "{date}_{name}" --date-format "%Y%m%d" --no-hidden

# 3. 清理空格 + 改扩展名，排除临时文件
python ren.py ./doc --squeeze --ext .md -x "~*"

# 4. 只想动一个文件：全局规则 + 单文件覆盖
python ren.py ./doc --prefix "v2_" --map "会议记录.md=2024-会议记录.md"

# 5. 存档对照表，两个版本的规则结果可以直接 diff
python ren.py ./doc --template "{parent}_{n:02d}_{name}" --csv 方案A.csv

# 6. 执行，然后后悔了
python ren.py ./doc --template "{parent}_{n:02d}_{name}" --apply
python ren.py --undo .ren-journal/ren-20250101-101010-ab12.json --apply
python ren.py --list-journals
```

## 几个刻意的取舍

**预览即执行结果**。规则只作用于内存里的 `rows`，`--apply` 只是把同一份 `rows` 交给 `execute()`，不存在两套逻辑。预览里红色是被替换掉的部分、绿色是新加的部分，改在哪一眼可见。

**先验证后动手，失败就整体回退**。`build()` 在预览阶段就算完了全部后果：空名、非法字符、Windows 保留名（CON/PRN/COM1…）、结尾空格点、255 字节文件名、路径长度临界（Windows 会在 240 字符时就给警告而不是等到 260 才炸）、目标已存在、批次内重名、大小写不敏感碰撞。有 error 就整批拒绝，绝不"跳过继续"。

**两阶段改名**。先全部挪到 `.ren-tmp-*`，再落到目标名，因此 `a→b, b→c` 这类链式和交换场景不会互相踩踏；中途任何异常按逆序回滚，回滚失败会逐条列出供手工处理。

**撤销是一等公民**，不是口头承诺。每次 `--apply` 都落 `.ren-journal/*.json`，记录绝对路径与 size/mtime；撤销同样走两阶段，且执行前先校验"当前文件还在、原位置没被占"，不满足就整批不动。

**单个文件仍可干预**。`-x` 按 glob 排除，`--map 旧名=新名` 精确覆盖某一条，`--map` 里写了目录里不存在的名字会报错而不是静默忽略。

**文件名生态按真实情况处理**。`nkey()` 让 `2.jpg` 排在 `10.jpg` 前面；宽度用 `east_asian_width` 算，中文名不会把表格撑歪；`.bashrc` 这类隐藏文件用点号计数特判，不会被 `splitext` 误拆成空 stem；`--ci-check auto` 在 Windows/macOS 上自动开启大小写碰撞检测，Linux 上关掉以免误报。

**常驻回归素材**。建议建一个 `samples/` 目录，固定放入：中文名、含空格、`.txt` 与 `.TXT` 并存、`CON.txt`、200 字符长名、只读文件、`~$临时.docx`。每次改动规则逻辑跑同一条命令，用 `--csv` 输出去 diff 上一次的对照表——对错看清单，不靠肉眼扫目录。
