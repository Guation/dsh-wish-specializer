function cr(p0,p1,p2,p3,t){
  const t2=t*t,t3=t2*t;
  return 0.5*(2*p1+(-p0+p2)*t+(2*p0-5*p1+4*p2-p3)*t2+(-p0+3*p1-3*p2+p3)*t3);
}
function sample(keys,n){
  const m=keys.length, out=[];
  for(let i=0;i<n;i++){
    const t=i/(n-1)*(m-1), i1=Math.min(m-2,Math.floor(t)), f=t-i1;
    const i0=Math.max(0,i1-1), i2=Math.min(m-1,i1+1), i3=Math.min(m-1,i1+2);
    const row=[];
    for(let j=0;j<6;j++)
      row.push(cr(keys[i0][j],keys[i1][j],keys[i2][j],keys[i3][j],f));
    out.push(row);
  }
  return out;
}

const ARCHES = [[1.40,0.35],[-1.45,0.35]];   // 轮拱中心
function inArch(x,y,z){
  if(Math.abs(z) < 0.55) return false;
  for(const [wx,wy] of ARCHES){
    const dx=x-wx, dy=y-wy;
    if(dx*dx+dy*dy < 0.44*0.44) return true;
  }
  return false;
}
function buildLoft(keys, seg=110, K=12, e=0.6, cutArch=false){
  const rows = sample(keys, seg), M = 4*K, pos = [];
  for(let i=0;i<seg;i++){
    const r = rows[i], h = [];
    for(let k=0;k<=K;k++){                       // 下段：底心 → 最宽
      const a = k/K*Math.PI/2;
      h.push([r[4]*Math.pow(Math.sin(a),e), r[2]-(r[2]-r[1])*Math.pow(Math.cos(a),e)]);
    }
    for(let k=1;k<=K;k++){                       // 上段：最宽 → 顶心
      const b = k/K*Math.PI/2;
      h.push([r[5]+(r[4]-r[5])*Math.pow(Math.cos(b),e),
              r[2]+(r[3]-r[2])*Math.pow(Math.sin(b),e)]);
    }
    const ring=[];
    for(let k=0;k<h.length;k++) ring.push([ h[k][0], h[k][1]]);
    for(let k=h.length-2;k>=1;k--) ring.push([-h[k][0], h[k][1]]);
    for(const p of ring) pos.push(r[0], p[1], p[0]);
  }
  const idx=[];
  const P=(i,j)=>[pos[(i*M+j)*3],pos[(i*M+j)*3+1],pos[(i*M+j)*3+2]];
  for(let i=0;i<seg-1;i++) for(let j=0;j<M;j++){
    const a=i*M+j, b=i*M+(j+1)%M, c=(i+1)*M+j, d=(i+1)*M+(j+1)%M;
    if(cutArch){
      const p=P(i,j);
      if(inArch(p[0],p[1],p[2])) continue;
    }
    idx.push(a,c,b, b,c,d);
  }
  const g=new THREE.BufferGeometry();
  g.setAttribute('position', new THREE.Float32BufferAttribute(pos,3));
  g.setIndex(idx); g.computeVertexNormals();
  return g;
}

function makeWheel(r,w){
  const g = new THREE.Group();            // 自转组，绕 z 轴滚动
  const add = (geo,mat)=>{
    const m = new THREE.Mesh(geo,mat); m.rotation.x = Math.PI/2;

}

function update(dt){
  const acc  = keys.KeyW || keys.ArrowUp;
  const brk  = keys.KeyS || keys.ArrowDown;
  const left = keys.KeyA || keys.ArrowLeft;
  const right= keys.KeyD || keys.ArrowRight;

  if(acc)  speed += (14*(1 - speed/VMAX) + 1.5) * dt;
  if(brk)  speed  = Math.max(0, speed - 26*dt);
  if(keys.Space) speed *= Math.max(0, 1 - 2.4*dt);
  speed = clamp(speed - speed*0.10*dt, 0, VMAX);

  // 速度越高，可用转角越小（稳定的高速手感）
  const maxSteer = 0.34 * (1 - Math.min(1, speed/70)*0.72);
  const target = (left?1:0) - (right?1:0);
  steerAng += (target*maxSteer - steerAng) * Math.min(1, dt*9);
  yaw += (speed / 2.85) * Math.tan(steerAng) * dt;

  car.position.x += Math.cos(yaw) * speed * dt;
  car.position.z -= Math.sin(yaw) * speed * dt;
  car.rotation.y = yaw;

  // 车身姿态
  roll  += (-steerAng*speed*0.045 - roll)  * Math.min(1, dt*6);
  const pT = acc ? -0.022 : (brk ? 0.03 : 0);
  pitch += (pT - pitch) * Math.min(1, dt*5);
  tilt.rotation.z = roll;
  tilt.rotation.x = pitch;

  // 车轮
  for(const w of wheels){
    if(w.front) w.steer.rotation.y = steerAng;
    w.spin.rotation.z -= speed * dt / w.r;
  }

  // 相机
  const fwd = new THREE.Vector3(Math.cos(yaw), 0, -Math.sin(yaw));
  let target_pos, target_look;
  if(camMode === 0){
    target_pos  = car.position.clone().addScaledVector(fwd, -6.4).setY(1.95);
    target_look = car.position.clone().addScaledVector(fwd, 3.2).setY(0.65);
  }else{
    cockpit.visible = false;
    target_pos  = car.position.clone().addScaledVector(fwd, -0.45).setY(1.02);
    target_look = car.position.clone().addScaledVector(fwd, 12).setY(0.95);
  }
  if(camMode === 0) cockpit.visible = true;
  const k = camMode === 0 ? Math.min(1, dt*7) : 1;
  camera.position.lerp(target_pos, k);
  camLook.lerp(target_look, k);
  camera.lookAt(camLook);

  // 主光跟随
  key.position.set(car.position.x + 8, 15, car.position.z + 9);
  key.target.position.copy(car.position);
  key.target.updateMatrixWorld();

  gearEl.textContent = speed < 0.5 ? 'N'
    : speed < 18 ? '1' : speed < 32 ? '2' : speed < 46 ? '3' : speed < 62 ? '4' : '5';
}

### 驾驶操作与视觉体验

您可以用键盘控制跑车，仪表盘和路面反馈会即时跟上。

- **驾驶操控**：W/↑加速、S/↓刹车、A/D转向，空格手刹。速度提升会自然降低转向灵敏度，高速驾驶更稳。
- **视角与外观**：按C键切换第三人称追尾和第一人称车内视角；数字键1、2、3分别切换红、银、蓝三种金属车漆。
- **实时反馈**：左下角速度表、进度条和右侧挡位数字随车速变化，车身在转向或刹车时会有侧倾与俯仰，路边灯柱提供速度参照。
- **视觉风格**：车身采用金属清漆材质，轮毂五辐设计，配合车底灯光和阴影，强化跑车低趴宽体的姿态。
---

**优化建议：** 想调驾驶手感，可搜索代码中的VMAX（当前83）和tilt.rotation里的系数0.045、0.022、0.03。车漆颜色可替换paint相关色值。
